# prepare -> update inside the VM -> re-lock -> destroy clones -> rebuild template.

# Update-ClaudeBase.ps1 performs the prepare half only. Everything from
# "MAINTENANCE MODE STOP" onward is manual.

## MAINTENANCE MODE START ##
# Host: run from elevated PowerShell
Set-ExecutionPolicy -Scope Process Bypass
C:\VMs\Update-ClaudeBase.ps1 -Force -NoConnect

# Host: same, but also update the guest automatically over SSH.
# -AutoUpdateGuest waits for port 22, then runs ONE sudo session that does:
#   - apt update / upgrade / autoremove / clean
#   - npm install -g @anthropic-ai/claude-code@<-ClaudeCodeVersion>  (root global)
#   - npm install -g @openai/codex as the sandbox user
#   - copies C:\VMs\SharedFolder\ contents into /home/sandbox, then chowns them
#   - prints claude and codex versions
# It logs in as 'user' and sudo prompts once, interactively.
# No password is stored in the script or on disk.
C:\VMs\Update-ClaudeBase.ps1 -Force -NoConnect -AutoUpdateGuest

# Relevant defaults, override only if something has moved:
#   -BaseVmIp           172.30.101.50
#   -SshUser            user
#   -SshKeyPath         C:\VMs\ssh\claude_sandbox_ed25519
#   -SandboxUser        sandbox
#   -ClaudeCodeVersion  2.1.258
#   -SharedFolderPath   C:\VMs\SharedFolder
#   -SshTimeoutSeconds  180
C:\VMs\Update-ClaudeBase.ps1 -AutoUpdateGuest -ClaudeCodeVersion 2.1.263

# NOTE: the SharedFolder copy overwrites same-named files in /home/sandbox.
# The run prints "OVERWRITING:" or "adding:" per path before it copies.

# Host: confirm maintenance networking is actually active before connecting.
# There must be NO "0.0.0.0/0 Deny" row. If there is, apply it manually.
C:\VMs\Set-ClaudeVMNetworkMode.ps1 -Mode status -VMName claude-base
C:\VMs\Set-ClaudeVMNetworkMode.ps1 -Mode maintenance -VMName claude-base

# Connect after the script starts claude-base.
# 'user' is the administrative account used for sudo work.
# 'sandbox' is the workload account that owns /home/sandbox and runs codex.
ssh -i "C:\VMs\ssh\claude_sandbox_ed25519" user@172.30.101.50
ssh -i "C:\VMs\ssh\claude_sandbox_ed25519" sandbox@172.30.101.50

# The steps below are the manual equivalent of -AutoUpdateGuest.
# Skip them if the script already did the guest update.

# Inside claude-base: verify maintenance networking
sudo apt update
sudo apt upgrade -y
sudo apt autoremove -y
sudo apt clean

# Inside claude-base: make any baseline changes future clones should inherit
# Claude Code is a root-owned npm global install (readme section 13). Pin the
# version so the image is reproducible:
# sudo npm install -g @anthropic-ai/claude-code@2.1.258
# claude --version
# Other examples:
# pipx upgrade-all
# npm update -g
# docker pull <image>
# systemctl status ssh

## MAINTENANCE MODE STOP ##
# Host: when maintenance is finished, re-lock networking
C:\VMs\Set-ClaudeVMNetworkMode.ps1 -Mode locked -VMName claude-base

# Host: verify locked ACLs are restored
C:\VMs\Set-ClaudeVMNetworkMode.ps1 -Mode status -VMName claude-base

# Host: shut down claude-base cleanly
Stop-VM -Name "claude-base"

# Host: wait until the VM is fully off before copying the disk
while ((Get-VM -Name "claude-base").State -ne "Off") {
    Start-Sleep -Seconds 2
}

########
# Host: DESTROY ephemeral VMs before rebuilding the template
########
# Ephemeral differencing disks are parented to the template file. Replacing it
# breaks the parent linkage and leaves those VMs unbootable, so destroy them
# first. Update-ClaudeBase.ps1 only stops them, it does not remove them.
Get-VM | Where-Object Name -like "claude-ephemeral-*"

Get-VM | Where-Object Name -like "claude-ephemeral-*" | ForEach-Object {
    C:\VMs\Remove-ClaudeEphemeral.ps1 -Name $_.Name
}

########
# Host: REBUILD the read-only template from the updated base disk
########
# Skipping this step leaves every new clone on the previous image.
$template = "C:\VMs\templates\claude-base-template.vhdx"
$source   = "C:\HyperV\claude-base\claude-base.vhdx"

if (Test-Path $template) {
    Set-ItemProperty -Path $template -Name IsReadOnly -Value $false
    Remove-Item $template -Force
}

Copy-Item $source $template
Set-ItemProperty -Path $template -Name IsReadOnly -Value $true

# Host: final sanity checks
Get-Item $template, $source | Select-Object FullName, IsReadOnly
Get-VM -Name "claude-base"
Get-VM | Where-Object Name -like "claude-ephemeral-*"
C:\VMs\Set-ClaudeVMNetworkMode.ps1 -Mode status -VMName claude-base

# ### ###
# ERROR: Cannot connect to Claude in Sandbox
# ### ###
# SOLUTION:
# - Check locked ACLs:
#   C:\VMs\Set-ClaudeVMNetworkMode.ps1 -Mode status -VMName <vm-name>
# - Temporarily unlock networking:
#   C:\VMs\Set-ClaudeVMNetworkMode.ps1 -Mode maintenance -VMName <vm-name>
# - Re-lock when finished:
#   C:\VMs\Set-ClaudeVMNetworkMode.ps1 -Mode locked -VMName <vm-name>

# ### ###
# ERROR: IP access works but names do not resolve. sudo apt update fails.
# ### ###
# SYMPTOMS (inside the VM, in maintenance mode):
#   ping -c3 1.1.1.1                                  -> works
#   curl -4 -I --noproxy '*' http://1.1.1.1           -> works
#   getent hosts archive.ubuntu.com                   -> no output
#   sudo apt update / npm / curl by hostname          -> fail or time out
#
# CAUSE: This is a DNS problem, not an ACL problem. Hyper-V port ACLs are
#   address-only and cannot filter by port, so a working IP path alongside a
#   dead DNS path is never the lock script's fault. Some networks block
#   outbound port 53 to third-party resolvers and force resolution through
#   their own. Netplan ships with 1.1.1.1 and 8.8.8.8, which are unreachable
#   on such a network, so systemd-resolved has no working upstream.
#
# CONFIRM (Windows host): test the resolver the VM is configured to use.
#   Test-NetConnection -ComputerName 1.1.1.1 -Port 53
#   PingSucceeded True with TcpTestSucceeded False means port 53 is filtered
#   for the host as well, and the VM was never the problem.
#
# SOLUTION: point the VM at a resolver this network permits, usually its
#   gateway. Find the host's own resolver (elevated PowerShell):
#     Get-DnsClientServerAddress -AddressFamily IPv4 | Where-Object ServerAddresses
#
#   Test it in the VM before editing any config. This is runtime only:
#     sudo resolvectl dns eth0 <network-resolver>
#     resolvectl status
#     getent hosts archive.ubuntu.com
#
#   If that resolves, persist it in netplan:
#     sudo cp /etc/netplan/01-ai-sandbox.yaml /etc/netplan/01-ai-sandbox.yaml.bak
#     sudo nano /etc/netplan/01-ai-sandbox.yaml
#
#   Set the nameservers block, most specific resolver first. Indentation must
#   match the existing entries exactly:
#
#       nameservers:
#         addresses:
#           - <network-resolver>
#           - 1.1.1.1
#           - 8.8.8.8
#
#     sudo chmod 600 /etc/netplan/01-ai-sandbox.yaml
#     sudo netplan generate
#     sudo netplan apply
#     resolvectl status
#     getent hosts archive.ubuntu.com
#     ip -4 addr show eth0
#
#   Roll back with:
#     sudo cp /etc/netplan/01-ai-sandbox.yaml.bak /etc/netplan/01-ai-sandbox.yaml
#     sudo netplan apply
#
# NOTES:
# - systemd-resolved tries the list in order and fails over on timeout, so one
#   config covers both permissive and restricted networks. The cost is a few
#   seconds of timeout when the first entry is not present on the network.
# - The resolver is site-specific. Working from a different network may need a
#   different first entry.
# - resolvectl dns is lost on reboot and on the next netplan apply. Only the
#   netplan edit persists.
# - The fix reaches ephemeral clones only after the template is rebuilt.
# - Locked mode blocks port 53 by design, so this only ever affects maintenance
#   windows. Claude hostnames stay pinned in /etc/hosts (readme section 15).
# - While DNS is working, re-verify those pinned addresses. This is the only
#   point in the cycle where real DNS is available:
#     dig +short api.anthropic.com platform.claude.com claude.ai
#   Each should fall inside 160.79.104.0/21.
# - Permanent alternative: resolve over DNS-over-TLS (TCP 853) or
#   DNS-over-HTTPS (TCP 443) so no resolver address is configured and no
#   network's DNS policy applies. Requires allowlisting the resolver address
#   for locked mode, which is a deliberate loosening of the current posture.
